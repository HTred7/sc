    b.Font = Enum.Font.Gotham
    b.TextSize = 13
    b.Parent = parent
    round(b, 6)
    return b
end

-- FOV CHANGER
section("FOV CHANGER")

local currentFOV = 70
local camera = workspace.CurrentCamera

if camera then
    currentFOV = camera.FieldOfView
end

local fovLabel = Instance.new("TextLabel")
fovLabel.Size = UDim2.new(1, 0, 0, 30)
fovLabel.BackgroundTransparency = 1
fovLabel.TextColor3 = Color3.new(1, 1, 1)
fovLabel.Font = Enum.Font.GothamBold
fovLabel.TextSize = 15
fovLabel.Parent = content

local function updateFOV()
    camera = workspace.CurrentCamera
    if camera then
        camera.FieldOfView = currentFOV
    end
    fovLabel.Text = "FOV: " .. math.floor(currentFOV)
end

updateFOV()

action("Aumentar FOV (+5)", function()
    currentFOV = math.clamp(currentFOV + 5, 40, 120)
    updateFOV()
end)

action("Diminuir FOV (-5)", function()
    currentFOV = math.clamp(currentFOV - 5, 40, 120)
    updateFOV()
end)

action("Restaurar FOV (70)", function()
    currentFOV = 70
    updateFOV()
end)

workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(updateFOV)

-- SKIN CHANGER LOCAL
section("SKIN CHANGER")

local skinContainer = Instance.new("Frame")
skinContainer.Size = UDim2.new(1, 0, 0, 0)
skinContainer.AutomaticSize = Enum.AutomaticSize.Y
skinContainer.BackgroundTransparency = 1
skinContainer.Parent = content

local skinLayout = Instance.new("UIListLayout")
skinLayout.Padding = UDim.new(0, 6)
skinLayout.Parent = skinContainer

local function clearSkin(character)
    for _, item in ipairs(character:GetChildren()) do
        if item:GetAttribute("PanelSkin") then
            item:Destroy()
        end
    end
end

local function equipSkin(skin)
    local character = player.Character
    if not character then return end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    clearSkin(character)

    for _, item in ipairs(skin:GetChildren()) do
        if item:IsA("Accessory") then
            local clone = item:Clone()
            clone:SetAttribute("PanelSkin", true)
            humanoid:AddAccessory(clone)

        elseif item:IsA("Shirt")
            or item:IsA("Pants")
            or item:IsA("ShirtGraphic") then

            local old = character:FindFirstChildOfClass(item.ClassName)
            if old then old:Destroy() end

            local clone = item:Clone()
            clone:SetAttribute("PanelSkin", true)
            clone.Parent = character
        end
    end
end

local function refreshSkins()
    for _, child in ipairs(skinContainer:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end

    if not skinsFolder then
        local info = button(skinContainer, "Pasta Skins nao encontrada")
        info.Active = false
        return
    end

    local skins = skinsFolder:GetChildren()

    table.sort(skins, function(a, b)
        return a.Name < b.Name
    end)

    for _, skin in ipairs(skins) do
        if skin:IsA("Folder") then
            local b = button(skinContainer, "Equipar: " .. skin.Name)
            b.MouseButton1Click:Connect(function()
                equipSkin(skin)
            end)
        end
    end
end

refreshSkins()

if skinsFolder then
    skinsFolder.ChildAdded:Connect(refreshSkins)
    skinsFolder.ChildRemoved:Connect(refreshSkins)
end

-- ARRASTAR PAINEL
local dragging = false
local dragStart
local startPosition
local dragInput

title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

title.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and input == dragInput then
        local delta = input.Position - dragStart

        main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)
